2. Training Optimization
-----------------------

2.1 Batch Size Optimization
```python
# Adjust batch size based on GPU memory
batch_size = 64  # Start with this value
```

2.2 Memory Management
```python
# Clear GPU cache before training
torch.cuda.empty_cache()

# Use gradient checkpointing for large models
model.use_checkpointing = True

# Monitor GPU memory usage
def print_gpu_memory():
    if torch.cuda.is_available():
        print(f"GPU Memory Usage:")
```


4.2 Training Speed
```python
# Measure training speed
import time

def measure_training_speed(model, data_loader):
    model.eval()
    times = []
    with torch.no_grad():
        for batch in data_loader:
            start_time = time.time()
            _ = model(batch['llrs'])
            times.append(time.time() - start_time)
    return sum(times) / len(times)
```

7. Example Training Script
-------------------------

```python
import torch
from torch.cuda.amp import autocast, GradScaler

def train_with_gpu(model, train_loader, optimizer, num_epochs):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = model.to(device)
    scaler = GradScaler()
    
    for epoch in range(num_epochs):
        model.train()
        total_loss = 0
        
        for batch in train_loader:
            # Move data to GPU
            batch = {k: v.to(device) for k, v in batch.items()}
            
            # Forward pass with mixed precision
            with autocast():
                outputs = model(batch['llrs'])
                loss = compute_loss(outputs, batch['bits'], batch['syndrome'])
            
            # Backward pass with gradient scaling
            optimizer.zero_grad()
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            
            total_loss += loss.item()
            
            # Monitor GPU memory
            if torch.cuda.is_available():
                print_gpu_memory()
        
        avg_loss = total_loss / len(train_loader)
        print(f"Epoch {epoch+1}, Average Loss: {avg_loss:.4f}")
```
